# Docker Compose interpolates ${VAR} only inside docker-compose.yml, NOT

# within .env values — so every value below must be a literal, there is no

# $REF indirection. The LORE_* values shown are exactly the compose

# defaults, so with Ollama running locally you can skip this file entirely

# (no API keys required).

# ── Optional API keys — only needed for cloud models ─────────────────

# Uncomment and fill in only if you use Claude / DeepSeek / GPT-4o.

# An unedited copy (sk-... left as-is) sends a fake key to those providers.

#

# DEEPSEEK_API_KEY is the canonical name. OPENAI_API_KEY is the older name

# from when this stack was OpenAI-first; docker-compose.yml falls back from

# each to the other, so setting either one works. DEEPSEEK_API_KEY wins if

# both are set. Only one is needed.

# DEEPSEEK_API_KEY=sk-...

# OPENAI_API_KEY=sk-...

# ANTHROPIC_API_KEY=sk-ant-...

# ── Lore — defaults shown; edit only if you must override ────────────

LORE_UPSTREAM_OPENAI=http://litellm:4000
LORE_UPSTREAM_ANTHROPIC=http://litellm:4000
LORE_WORKER_UPSTREAM=http://litellm:4000
LORE_WORKER_MODEL=openai/qwen3:8b
LORE_WORKER_API_KEY=sk-litellm-master
LORE_DEBUG=true

# LORE_WORKER_MODEL is "provider/model": the part after the slash is the

# model name sent upstream and MUST match a model_name in config/litellm.yaml

# (qwen3:8b, deepseek-flash, ...). The part before the slash is the

# provider ID, which selects the protocol:

# openai -> OpenAI chat completions (LiteLLM, DeepSeek, Ollama)

# anthropic -> Anthropic Messages API

# The worker does not parse a prefix string; it resolves providerID and derives

# the protocol from that. Workers MUST use the same provider as the session:

# cross-provider worker calls fail (wrong credentials, wrong API format).

#

# The LORE_UPSTREAM_* URLs above are server ROOTS -- no trailing /v1. Lore

# appends the API path itself (its gateway builds `${upstreamAnthropic}/v1/models`),

# so a /v1 suffix produces /v1/v1/messages (404) and /v1/v1/files (400).
